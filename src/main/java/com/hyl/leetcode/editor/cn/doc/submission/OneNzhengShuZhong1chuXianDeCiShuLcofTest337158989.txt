class Solution {
    public int countDigitOne(int n) {
        int sum = 0;
        for (int i=1;i<=n;i++){
            sum +=read(i);
        }
        return sum;
    }

    public int read(int n){
        int count =0;
        while(n>0){
            if (n%10==1){
                count++;
            }
            n = n/10;
        }
        return count;
    }

}
