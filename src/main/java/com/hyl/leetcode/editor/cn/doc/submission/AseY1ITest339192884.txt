class Solution {
    public int maxProduct(String[] words) {
        int res = 0;
        int len = words.length;
        int[] bit = new int[len];
        for (int i=0;i<len;i++){
            for (int j=0;j<words[i].length();j++){
                bit[i] |= 1<<(words[i].charAt(j)-'a');
            }
        }
        for (int i=0;i<len;i++){
            for (int j=i+1;j<len;j++){
                if ((bit[i]&bit[j])==0){
                    res = Math.max(res,words[i].length()*words[j].length());
                }
            }
        }
        return res;
    }


}
