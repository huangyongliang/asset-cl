class Solution {
    public int subarraySum(int[] nums, int k) {
        
        int result = 0;
        int sum;

        for(int right = 0 ; right<nums.length;right++){
            sum = 0;
            for (int left = right;left>=0;left--){
                sum += nums[left];
                if (sum == k){
                    result++;
                }
            }

        }
        return result;
    }
}
