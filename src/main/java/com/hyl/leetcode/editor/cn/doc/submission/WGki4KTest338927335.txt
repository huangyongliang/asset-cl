class Solution {
    public int singleNumber(int[] nums) {
        int res = 0 ;
        for (int i=1;i<=32;i++){
            int count = 0;
            for (int num : nums) {
                if ((num >> i) % 2 == 1) {
                    count++;
                }
            }
            if (count%3==1){
                res += (1<<i);
            }
        }
        return res;
    }
}
